import os from openai import OpenAI # 1. 소크라테스 튜터 시스템 프롬프트 정의 SOCRATES_SYSTEM_PROMPT = """ [역할 정의] 너는 학습자의 스스로 사고하는 능력을 길러주는 친절하고 지혜로운 '소크라테스 튜터'이다. 너의 목표는 정답을 직접적으로 제공하는 것이 아니라, 대화와 역질문, 힌트를 통해 학습자가 스스로 문제의 답을 찾아가고 생각을 확장할 수 있도록 돕는 것이다. [기본 원칙] 1. 절대로 정답을 바로 말하지 않는다. (문제의 정답, 공식, 해결책 직접 제시 금지) 2. 질문에는 '역질문'으로 응답한다. 3. 단계별 힌트를 제공한다. 4. 사고의 확장을 유도한다. (정답 도달 시 이유나 타 사례 적용 질문) 5. 공감과 칭찬을 아끼지 않는다. [대화 진행 흐름] 1. 파악하기 -> 2. 안내하기 (역질문 & 힌트) -> 3. 검증하기 -> 4. 확장하기 [말투 및 톤앤매너] - 친절하고 다정하며 차분한 어조 (~해요, ~해볼까요? 등) - 학습자의 탐구심을 자극하는 존중하는 태도 """ # 2. 클라이언트 및 대화 기록 초기화 client = OpenAI(api_key=os.environ.get("OPENAI_API_KEY")) messages = [ {"role": "system", "content": SOCRATES_SYSTEM_PROMPT}, { "role": "assistant", "content": "안녕하세요! 반갑습니다. 여러분의 생각을 넓혀주는 탐구 여정의 동반자, 소크라테스 튜터입니다.\n\n어떤 문제나 주제에 대해 이야기를 나누고 싶으신가요? 먼저 생각해보신 부분부터 편하게 말씀해 주세요!", }, ] print(f"튜터: {messages[1]['content']}\n") # 3. 대화 루프 실행 while True: user_input = input("학습자: ") if user_input.strip().lower() in ["exit", "quit", "종료"]: print("\n튜터: 오늘 나누었던 탐구 과정이 멋진 자산이 되었길 바라요. 다음에 또 함께 고민해봐요!") break # 사용자 입력 추가 messages.append({"role": "user", "content": user_input}) # LLM 응답 생성 response = client.chat.completions.create( model="gpt-4o", # 필요에 따라 모델명을 변경할 수 있습니다. messages=messages, temperature=0.7, ) bot_reply = response.choices[0].message.content # 답변 추가 및 출력 messages.append({"role": "assistant", "content": bot_reply}) print(f"\n튜터: {bot_reply}\n")
